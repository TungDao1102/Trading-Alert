# ---- memory guard (Windows, ctypes; no psutil) -----------------------------------------------
class _PMC(ctypes.Structure):
    _fields_ = [("cb", wintypes.DWORD), ("PageFaultCount", wintypes.DWORD),
                ("PeakWorkingSetSize", ctypes.c_size_t), ("WorkingSetSize", ctypes.c_size_t),
                ("QuotaPeakPagedPoolUsage", ctypes.c_size_t), ("QuotaPagedPoolUsage", ctypes.c_size_t),
                ("QuotaPeakNonPagedPoolUsage", ctypes.c_size_t), ("QuotaNonPagedPoolUsage", ctypes.c_size_t),
                ("PagefileUsage", ctypes.c_size_t), ("PeakPagefileUsage", ctypes.c_size_t)]
class _MS(ctypes.Structure):
    _fields_ = [("dwLength", wintypes.DWORD), ("dwMemoryLoad", wintypes.DWORD),
                ("ullTotalPhys", ctypes.c_ulonglong), ("ullAvailPhys", ctypes.c_ulonglong),
                ("ullTotalPageFile", ctypes.c_ulonglong), ("ullAvailPageFile", ctypes.c_ulonglong),
                ("ullTotalVirtual", ctypes.c_ulonglong), ("ullAvailVirtual", ctypes.c_ulonglong),
                ("ullAvailExtendedVirtual", ctypes.c_ulonglong)]
_k32 = ctypes.WinDLL("kernel32", use_last_error=True); _psapi = ctypes.WinDLL("psapi", use_last_error=True)
_k32.GetCurrentProcess.restype = wintypes.HANDLE
_psapi.GetProcessMemoryInfo.argtypes = [wintypes.HANDLE, ctypes.POINTER(_PMC), wintypes.DWORD]
GIB = 1024 ** 3
WS_LIMIT = 3 * GIB          # abort when this process' working set exceeds 3 GiB
AVAIL_FLOOR = 1 * GIB       # abort when system available physical memory drops below 1 GiB
def _mem():
    pmc = _PMC(); pmc.cb = ctypes.sizeof(pmc)
    _psapi.GetProcessMemoryInfo(_k32.GetCurrentProcess(), ctypes.byref(pmc), pmc.cb)
    ms = _MS(); ms.dwLength = ctypes.sizeof(ms); _k32.GlobalMemoryStatusEx(ctypes.byref(ms))
    return pmc, ms
_state = {"phase": "start", "peak_ws": 0, "min_avail": 1 << 62, "split_peak": 0, "split_min_avail": 1 << 62}
def _guard():
    while True:
        pmc, ms = _mem()
        _state["peak_ws"] = max(_state["peak_ws"], pmc.WorkingSetSize, pmc.PeakWorkingSetSize)
        _state["split_peak"] = max(_state["split_peak"], pmc.WorkingSetSize)
        _state["min_avail"] = min(_state["min_avail"], ms.ullAvailPhys)
        _state["split_min_avail"] = min(_state["split_min_avail"], ms.ullAvailPhys)
        if pmc.WorkingSetSize > WS_LIMIT or ms.ullAvailPhys < AVAIL_FLOOR:
            print(f"MEMORY_ABORT\tphase={_state['phase']}\tworking_set_gib={pmc.WorkingSetSize/GIB:.2f}"
                  f"\tavail_gib={ms.ullAvailPhys/GIB:.2f}\tpeak_ws_gib={_state['peak_ws']/GIB:.2f}"
                  f"\telapsed_s={time.perf_counter()-T0:.0f}", flush=True)
            os._exit(3)
        time.sleep(0.5)
_pmc0, _ms0 = _mem()
print(f"START\tavail_gib={_ms0.ullAvailPhys/GIB:.2f}\ttotal_gib={_ms0.ullTotalPhys/GIB:.2f}", flush=True)
threading.Thread(target=_guard, daemon=True).start()
def phase(name):
    _state["phase"] = name
    pmc, ms = _mem()
    _state["split_peak"] = max(_state["split_peak"], pmc.WorkingSetSize)
    print(f"PHASE\t{name}\telapsed_s={time.perf_counter()-T0:.1f}\tws_gib={pmc.WorkingSetSize/GIB:.2f}"
          f"\tpeak_ws_gib={max(pmc.PeakWorkingSetSize,_state['peak_ws'])/GIB:.2f}"
          f"\tavail_gib={ms.ullAvailPhys/GIB:.2f}", flush=True)

# ---- Holdout guard on the loader: every parquet read and every dataset-id hash -------------
_reads = {"parquet": set(), "hashed": set()}
def _assert_pre_holdout(path, what):
    m = _DATE_RE.search(str(path).replace("\\", "/"))
    if m is None or m.group(1) >= HOLDOUT_DAY:
        raise AssertionError(f"HOLDOUT_GUARD: refusing to {what} {path}")
_orig_read_parquet = pl.read_parquet
def _guarded_read_parquet(source, *a, **k):
    _assert_pre_holdout(source, "read")
    _reads["parquet"].add(str(source))
    return _orig_read_parquet(source, *a, **k)
pl.read_parquet = _guarded_read_parquet
_orig_file_sha = ci_mod._file_sha256
def _guarded_file_sha(path):
    _assert_pre_holdout(path, "hash")
    _reads["hashed"].add(str(path))
    return _orig_file_sha(path)
ci_mod._file_sha256 = _guarded_file_sha
