files = {}
checked = 0
for line in inv_bytes.decode("utf-8").splitlines():
    h, rel = line.split("  ", 1)
    m = re.search(r"date=(\d{4}-\d{2}-\d{2})", rel)
    if m and m.group(1) < "2026-05-10":
        files[rel] = h
    if f"symbol={SYM}/" in rel:
        got = hashlib.sha256((RAW / rel).read_bytes()).hexdigest()
        assert got == h, f"inventory mismatch {rel}"
        checked += 1
ds = sha256_hex({"files": [{"path": p, "sha256": files[p]} for p in sorted(files)]})
assert ds == DATASET_ID, f"dataset id from inventory {ds}"
print(f"PRECONDITIONS OK: S0 sha256 verified; listing sha256 verified; dataset id {DATASET_ID} "

# 3. 48 points ---------------------------------------------------------------
points = []
for g in sorted(glob.glob(str(REPO / "docs/research/preregistration/research-run-1-grid-*.json"))):
    grid = load_grid(Path(g))
    points += [(grid, p) for p in grid.points]
assert len(points) == 48, len(points)
_only = os.environ.get("REEF_ONLY")
if _only:
    points = [(g, p) for g, p in points if p.parameter_set_id == _only]
    assert len(points) == 1, _only

def iso(t): return None if t is None else t.isoformat().replace("+00:00", "Z")
def cnt(items):
    d = {}
    for i in items: d[str(i)] = d.get(str(i), 0) + 1
    return dict(sorted(d.items()))

for grid, point in points:
    pid = point.parameter_set_id
    ps = load_parameter_set(grid.strategy_id, point.parameter_set_id)
    assert ps.parameter_set_hash == point.parameter_set_hash
    cost = {k: str(v) for k, v in cost_slice(ps.params).items()}
    try:
        data = build()
        case = data.to_case_input(
            strategy_id=grid.strategy_id, strategy_version=ps.strategy_version,
            parameter_set_id=pid, account_state={"equity": str(manifest.equity), "risk_rate": str(manifest.risk_rate)},
            cost_model=cost, constraint_history=constraints, terminal_settlements=case_rows)
        last_funding = max(datetime.fromisoformat(r["funding_time_utc"]) for r in case["funding"]
                           if r["symbol"] == SYM)
        src = ParquetMinuteSource(RAW, symbols=[SYM], start_utc=dev_window.start_utc, end_utc=dev_window.end_utc)
        out = simulator.run(case, minute_source=src)
    except BaseException as exc:
        tb = traceback.extract_tb(exc.__traceback__)[-1]
        print(f"{pid}\tERROR\t{type(exc).__name__}\t{exc}\t@ {Path(tb.filename).name}:{tb.lineno}")
        continue
    tr = out["trades"]; rs = out["run_summary"]
    funding_rows = [datetime.fromisoformat(r["funding_time_utc"]) for r in case["funding"] if r["symbol"] == SYM]
    settle, cancel, pend, after830, equal_bound_other, mark_missing_on_settlement = [], [], [], [], [], 0
    for t in tr:
        et = datetime.fromisoformat(t["entry_time_utc"]) if t["entry_time_utc"] else None
        xt = datetime.fromisoformat(t["exit_time_utc"]) if t["exit_time_utc"] else None
        if et is not None and et > NO_NEW: after830.append(f"entry={iso(et)} exit_kind={t['exit_kind']}")
        if t["entry_kind"] == "PENDING_AT_DATA_END":
            pend.append(f"signal_close={t['signal_close_utc']} resolved_at={t['resolved_at_utc']}")
        if t["entry_kind"] == "SETTLEMENT_CANCEL":
            cancel.append(f"direction={t['direction']} signal_close={t['signal_close_utc']} "
                          f"resolved_at={t['resolved_at_utc']} entry_time={t['entry_time_utc']} "
                          f"exit_kind={t['exit_kind']}")
        if t["exit_kind"] == "SETTLEMENT":
            px = Decimal(t["exit_price"])
            side = "lower" if px == LOWER else "upper" if px == UPPER else "NEITHER"
            note = t["coverage_note"] or ""
            if "mark_price_1m_missing" in note: mark_missing_on_settlement += 1
            expect_side = "lower" if t["direction"] == "LONG" else "upper"
            exp_funding = sum(1 for f in funding_rows if et <= f < xt)
            settle.append(
                f"direction={t['direction']} bound_used={side} (expected_for_direction={expect_side}) "
                f"entry={iso(et)} exit={iso(xt)} exit_equals_T={xt == TERMINAL} "
                f"exit_equals_bound_exactly={px in (LOWER, UPPER)} "
                f"funding_event_count={t['funding_event_count']} funding_rows_entry_le_t_lt_exit={exp_funding} "
                f"note_prefix_ok={note.startswith('terminal_settlement') or 'terminal_settlement SETTLEMENT' in note} "
                f"mark_price_1m_missing_in_note={'mark_price_1m_missing' in note}")
        elif t["exit_price"] is not None and Decimal(t["exit_price"]) in (LOWER, UPPER):
            equal_bound_other.append(f"exit_kind={t['exit_kind']} exit_time={t['exit_time_utc']}")
    print(f"{pid}\tOK\tsemver={rs['simulator_semantics_version']}\tn_trades={len(tr)}"
          f"\texit_kinds={cnt([t['exit_kind'] for t in tr if t['exit_kind']])}"
          f"\tentry_kinds={cnt([t['entry_kind'] for t in tr])}"
          f"\tsettlement={settle or 'none'}\tsettlement_cancel={cancel or 'none'}"
          f"\tno_trade_counts={json.dumps(rs['no_trade_counts'], sort_keys=True)}"
          f"\tsettlement_count={rs.get('settlement_count', 'ABSENT')}"
          f"\tpending_at_data_end={pend or 'none'}"
          f"\tsummary(end_of_data={rs['end_of_data_count']},pending={rs['pending_at_data_end_count']},"
          f"allowed_gaps_used={len(rs.get('allowed_gaps_used', []))})"
          f"\tentry_after_0830={after830 or 'none'}"
          f"\tnon_settlement_exit_equals_a_bound={equal_bound_other or 'none'}"
          f"\tmark_price_1m_missing_on_settlement_notes={mark_missing_on_settlement}"
          f"\tlast_reef_funding_row={iso(last_funding)}")
    if settle:
        sample = next(t for t in tr if t["exit_kind"] == "SETTLEMENT")
        print(f"{pid}\tNOTE_SAMPLE\t{sample['coverage_note']}")
